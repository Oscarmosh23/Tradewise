// TradeWise Scalper Pro EA — Full 5-Layer + 1-Min Entry
extern string Sym = "XAUUSD";
extern double RiskPct = 0.5;
extern int StopPips = 10;
extern int TargetPips = 20;
extern int MinScore = 4; // 4=NORMAL, 5=STRONG
extern bool Use1MinEntry = true;

bool signalReady = false;
int signalDir = 0; // 1=buy, -1=sell

void OnTick(){
  if(OrdersTotal()>0){manageTrades();return;}
  int score = calcScore();
  if(score >= MinScore){signalReady=true;signalDir=1;}
  else if(score <= -MinScore){signalReady=true;signalDir=-1;}
  else{signalReady=false;}
  if(!signalReady) return;
  if(Use1MinEntry) checkM1Entry();
  else execute(signalDir);
}

void checkM1Entry(){
  double e8=iMA(Sym,PERIOD_M1,8,0,MODE_EMA,PRICE_CLOSE,0);
  double e21=iMA(Sym,PERIOD_M1,21,0,MODE_EMA,PRICE_CLOSE,0);
  double e8p=iMA(Sym,PERIOD_M1,8,0,MODE_EMA,PRICE_CLOSE,1);
  double e21p=iMA(Sym,PERIOD_M1,21,0,MODE_EMA,PRICE_CLOSE,1);
  bool freshBullCross=e8>e21&&e8p<=e21p;
  bool freshBearCross=e8=e21p;
  if(signalDir==1&&freshBullCross) execute(1);
  if(signalDir==-1&&freshBearCross) execute(-1);
}

int calcScore(){
  int s=0;
  double e8_5=iMA(Sym,PERIOD_M5,8,0,MODE_EMA,PRICE_CLOSE,0);
  double e21_5=iMA(Sym,PERIOD_M5,21,0,MODE_EMA,PRICE_CLOSE,0);
  double e8_15=iMA(Sym,PERIOD_M15,8,0,MODE_EMA,PRICE_CLOSE,0);
  double e21_15=iMA(Sym,PERIOD_M15,21,0,MODE_EMA,PRICE_CLOSE,0);
  double e8_1h=iMA(Sym,PERIOD_H1,8,0,MODE_EMA,PRICE_CLOSE,0);
  double e21_1h=iMA(Sym,PERIOD_H1,21,0,MODE_EMA,PRICE_CLOSE,0);
  double rsi5=iRSI(Sym,PERIOD_M5,14,PRICE_CLOSE,0);
  double rsi15=iRSI(Sym,PERIOD_M15,14,PRICE_CLOSE,0);
  double macdM=iMACD(Sym,PERIOD_M5,12,26,9,PRICE_CLOSE,MODE_MAIN,0);
  double macdS=iMACD(Sym,PERIOD_M5,12,26,9,PRICE_CLOSE,MODE_SIGNAL,0);
  if(e8_5>e21_5) s++; else s--;
  if(e8_15>e21_15) s++; else s--;
  if(rsi5<45) s++; else if(rsi5>55) s--;
  if(rsi15<45) s++; else if(rsi15>55) s--;
  if(e8_1h>e21_1h) s++; else s--;
  return s;
}

void manageTrades(){
  for(int i=0;i     if(OrderSelect(i,SELECT_BY_POS)&&OrderMagicNumber()==20260601){
      double profit=OrderProfit();
      double pipProfit=(OrderType()==OP_BUY)?
        (Bid-OrderOpenPrice())/(Point*10):
        (OrderOpenPrice()-Ask)/(Point*10);
      // Move to breakeven at +10 pips
      if(pipProfit>=10&&OrderStopLoss()!=OrderOpenPrice()){
        OrderModify(OrderTicket(),OrderOpenPrice(),OrderOpenPrice(),OrderTakeProfit(),0,clrYellow);
      }
    }
  }
}

void execute(int dir){
  double lots=calcLots();
  if(dir==1){
    double p=Ask,sl=p-StopPips*Point*10,tp=p+TargetPips*Point*10;
    OrderSend(Sym,OP_BUY,lots,p,3,sl,tp,"TWPro",20260601,0,clrGreen);
  }else{
    double p=Bid,sl=p+StopPips*Point*10,tp=p-TargetPips*Point*10;
    OrderSend(Sym,OP_SELL,lots,p,3,sl,tp,"TWPro",20260601,0,clrRed);
  }
}
double calcLots(){
  double r=AccountBalance()*RiskPct/100;
  double pv=MarketInfo(Sym,MODE_TICKVALUE)*10;
  return NormalizeDouble(r/(StopPips*pv),2);
}
