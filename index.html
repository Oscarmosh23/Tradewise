⚡ Accuracy Rules — How to Reach 75-80%
01
Only trade when Confluence Score is 5/6 or 6/6
The platform filters automatically. Below 5/6 it shows WAIT. You never need to guess — if it says STRONG BUY or STRONG SELL with 5+ layers passing, that is your setup. Anything less — sit on your hands.
02
Only trade during London or New York session
The session filter is now built into the signal. During Asian session the platform automatically reduces signal confidence by 15 points and marks it LOW CONFIDENCE. This eliminates most bad trades from thin market hours.
03
Volume must confirm — no volume = no trade
Layer 3 checks whether volume is above the 20-period average. If price is moving without volume it is likely a fake move that will reverse. The platform flags this automatically as a filter fail.
04
1-hour trend must agree with scalp direction
Layer 5 checks the 1-hour EMA direction. If you are scalping BUY on 5min but the 1-hour trend is DOWN — that is a counter-trend scalp with much lower probability. The platform marks this as a filter fail.
05
Confirm on MT4 before every entry
TradeWise gives the signal. MT4 gives visual confirmation. EMA 8 must be above EMA 21 on your 5min AND 15min chart. If both match — enter. If they do not match — wait for next signal.

🤖 MT4 EA — 6-Layer Agreement Logic
// TradeWise Scalper Pro EA — 6-Layer MTF Agreement
extern string Sym = "XAUUSD";
extern double RiskPct = 0.5;
extern int StopPips = 10;
extern int TargetPips = 20;
extern int MinScore = 5; // minimum layers to trade

void OnTick() {
  if(OrdersTotal() > 0) return;
  if(!IsTradeAllowed()) return;
  int score = CalcScore();
  if(score >= MinScore) ExecuteBuy();
  if(score <= -MinScore) ExecuteSell();
}

int CalcScore() {
  int s = 0;
  // Layer 1: EMA cross 5min
  double e8_5=iMA(Sym,PERIOD_M5,8,0,MODE_EMA,PRICE_CLOSE,0);
  double e21_5=iMA(Sym,PERIOD_M5,21,0,MODE_EMA,PRICE_CLOSE,0);
  if(e8_5>e21_5) s++; else s--;
  // Layer 2: EMA cross 15min
  double e8_15=iMA(Sym,PERIOD_M15,8,0,MODE_EMA,PRICE_CLOSE,0);
  double e21_15=iMA(Sym,PERIOD_M15,21,0,MODE_EMA,PRICE_CLOSE,0);
  if(e8_15>e21_15) s++; else s--;
  // Layer 3: RSI 5min
  double rsi5=iRSI(Sym,PERIOD_M5,14,PRICE_CLOSE,0);
  if(rsi5 < 45) s++; else if(rsi5 > 55) s--;
  // Layer 4: RSI 15min
  double rsi15=iRSI(Sym,PERIOD_M15,14,PRICE_CLOSE,0);
  if(rsi15 < 45) s++; else if(rsi15 > 55) s--;
  // Layer 5: 1-hour trend
  double e8_1h=iMA(Sym,PERIOD_H1,8,0,MODE_EMA,PRICE_CLOSE,0);
  double e21_1h=iMA(Sym,PERIOD_H1,21,0,MODE_EMA,PRICE_CLOSE,0);
  if(e8_1h>e21_1h) s++; else s--;
  // Layer 6: MACD 5min
  double macdMain=iMACD(Sym,PERIOD_M5,12,26,9,PRICE_CLOSE,MODE_MAIN,0);
  double macdSig=iMACD(Sym,PERIOD_M5,12,26,9,PRICE_CLOSE,MODE_SIGNAL,0);
  if(macdMain>macdSig) s++; else s--;
  return s;
}

void ExecuteBuy() {
  double price=Ask,sl=price-StopPips*Point*10,tp=price+TargetPips*Point*10;
  OrderSend(Sym,OP_BUY,CalcLots(StopPips),price,3,sl,tp,"TW6L",0,0,clrGreen);
}
void ExecuteSell() {
  double price=Bid,sl=price+StopPips*Point*10,tp=price-TargetPips*Point*10;
  OrderSend(Sym,OP_SELL,CalcLots(StopPips),price,3,sl,tp,"TW6L",0,0,clrRed);
}
double CalcLots(int sp) {
  double r=AccountBalance()*RiskPct/100;
  double pv=MarketInfo(Sym,MODE_TICKVALUE)*10;
  return NormalizeDouble(r/(sp*pv),2);
}
